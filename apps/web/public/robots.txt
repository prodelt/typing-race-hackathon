# Typing-Race. Everything a crawler may usefully index is public and needs no account: the
# product page, the formulas, the licences, the privacy statement, the about page, and the
# exercises themselves — any of them opens for a visitor with no account and no progress.
# What stays out is a learner's own state: a result is one person's attempt, and the session,
# settings and the old Today and Path addresses only make sense with a stored profile.
User-agent: *
Allow: /$
Allow: /formulas
Allow: /licences
Allow: /privacy
Allow: /about
Allow: /exercise/
Disallow: /today
Disallow: /path
Disallow: /result/
Disallow: /session
Disallow: /settings
